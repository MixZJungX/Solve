            $shopAccounts     = [];
            $externalAccounts = [];
            $notInDb          = [];

                if (isset($accountMap[$lower])) {
                    $acc = $accountMap[$lower];
                    if (($acc['account_type'] ?? 'shop') === 'shop') {
                        $shopAccounts[] = $acc;
                    } else {
                        $externalAccounts[] = $acc;
                    }
                } else {
                    $notInDb[] = $u;
                }
            }
